> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Скилл для Claude Code «в раздачу», который мониторит тренды в нише по RSS-фидам: недельный дайджест с горячими темами (через Claude), дешёвый пульс всплесков объёма (без Claude) и разовая визуальная HTML-карта, отчёты в Telegram.

**Architecture:** Зеркалит instawatch. `fetch.py` тянет фиды (feedparser) в SQLite и попутно копит частоты нормализованных терминов; `report.py` (weekly | pulse | map) читает базу и шлёт отчёт; `schedule.py` и `telegram.py` переносятся из instawatch почти без изменений. Секреты в `.env`, параметры в `config.json`.

**Tech Stack:** Python 3.9+, `feedparser`, `requests`, `pymorphy3` (лемматизация RU), SQLite (stdlib), headless `claude -p` для weekly/map, launchd/cron для расписания.

## Global Constraints

- Python 3.9+ — не использовать PEP 604 `X | Y` в аннотациях, только `Optional`/`Union` или без аннотаций (как в instawatch).
- Все тексты для пользователя — на русском.
- Секреты (`TELEGRAM_TOKEN`, `TELEGRAM_CHAT_ID`) только в `.env`, никогда в `config.json` и не в git.
- В каждой строке любого отчёта, где есть статья, обязана быть ссылка на неё.
- Лемматизация — только для кириллицы (pymorphy3); латиница — lowercase + стоп-слова. Если `pymorphy3` не установлен — фолбэк на исходный токен (без леммы), не падать.
- Все сравнения пульса — относительные (кратность базовой линии термина), пороги из `config.json`, ничего не хардкодить.
- Каждый Python-модуль запускается из папки скилла; пути строятся от `common.ROOT`.

# common.py
"""Общие утилиты rssradar: .env, конфиг, база, нормализация терминов."""
import json
import sqlite3
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parent
DATA_DIR = ROOT / "data"
DB_PATH = DATA_DIR / "rssradar.db"
CONFIG_PATH = ROOT / "config.json"
ENV_PATH = ROOT / ".env"

DEFAULTS = {
    "feeds": [],
    "pulse_min_feeds": 2,
    "pulse_spike_multiplier": 2.0,
    "pulse_baseline_days": 14,
    "pulse_window_days": 2,
    "pulse_cooldown_days": 5,
    "weekly_window_days": 7,
    "fetch_limit_cold": 30,
    "weekly_day": "sun",
    "weekly_time": "18:00",
    "pulse_time": "10:00",
}

SCHEMA = ""  # наполняется в Task 3


def load_env(path=None):
    path = Path(path or ENV_PATH)
    env = {}
    if not path.exists():
        return env
    for line in path.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        key, _, value = line.partition("=")
        value = value.strip().strip('"').strip("'")
        if value:
            env[key.strip()] = value
    return env


def load_config(path=None):
    path = Path(path or CONFIG_PATH)
    cfg = dict(DEFAULTS)
    if path.exists():
        cfg.update(json.loads(path.read_text(encoding="utf-8")))
    return cfg


def now_utc():
    return datetime.now(timezone.utc)


def parse_ts(value):
    """ISO-таймстамп → aware datetime. Терпит 'Z' и '+0000' (Python 3.9)."""
    value = value.replace("Z", "+00:00")
    if value.endswith("+0000"):
        value = value[:-5] + "+00:00"
    dt = datetime.fromisoformat(value)
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    return dt


def day_of(dt):
    return dt.astimezone(timezone.utc).strftime("%Y-%m-%d")


def connect(db_path=None):
    path = Path(db_path or DB_PATH)
    path.parent.mkdir(parents=True, exist_ok=True)
    con = sqlite3.connect(path)
    con.row_factory = sqlite3.Row
    if SCHEMA.strip():
        con.executescript(SCHEMA)
    return con
```

import re

try:
    import pymorphy3
    _MORPH = pymorphy3.MorphAnalyzer()
except Exception:          # noqa: BLE001 — нет pymorphy3 → фолбэк без лемм
    _MORPH = None

_TOKEN_RE = re.compile(r"[а-яёa-z0-9]+", re.IGNORECASE)
_RU_STOP = {
    "и", "в", "во", "не", "что", "он", "на", "я", "с", "со", "как", "а", "то",
    "все", "она", "так", "его", "но", "да", "ты", "к", "у", "же", "вы", "за",
    "бы", "по", "только", "ее", "мне", "было", "вот", "от", "меня", "о", "из",
    "для", "про", "это", "этот", "эта", "мы", "они", "или", "при", "над", "под",
}
_EN_STOP = {
    "the", "a", "an", "and", "or", "of", "to", "in", "on", "for", "is", "are",
    "was", "with", "as", "at", "by", "from", "how", "why", "what", "new", "your",
}


def normalize_terms(text):
    terms = []
    for tok in _TOKEN_RE.findall((text or "").lower()):
        if len(tok) < 3:
            continue
        if re.search(r"[а-яё]", tok):
            if tok in _RU_STOP:
                continue
            lemma = _MORPH.parse(tok)[0].normal_form if _MORPH else tok
        else:
            if tok in _EN_STOP:
                continue
            lemma = tok
        terms.append(lemma)
    return terms
```

from datetime import timedelta

SCHEMA = """
CREATE TABLE IF NOT EXISTS entries (
    feed_url   TEXT NOT NULL,
    entry_id   TEXT NOT NULL,
    title      TEXT,
    link       TEXT,
    published  TEXT,
    summary    TEXT,
    first_seen TEXT NOT NULL,
    PRIMARY KEY (feed_url, entry_id)
);
CREATE TABLE IF NOT EXISTS term_counts (
    term     TEXT NOT NULL,
    day      TEXT NOT NULL,
    feed_url TEXT NOT NULL,
    PRIMARY KEY (term, day, feed_url)
);
CREATE TABLE IF NOT EXISTS pulse_alerts (
    term       TEXT PRIMARY KEY,
    alerted_at TEXT NOT NULL
);
CREATE TABLE IF NOT EXISTS feed_status (
    feed_url  TEXT PRIMARY KEY,
    etag      TEXT,
    modified  TEXT,
    last_ok   TEXT,
    last_error TEXT
);
"""


def save_entries(con, records):
    new = 0
    stamp = now_utc().isoformat()
    for r in records:
        cur = con.execute(
            "INSERT OR IGNORE INTO entries "
            "(feed_url, entry_id, title, link, published, summary, first_seen) "
            "VALUES (?,?,?,?,?,?,?)",
            (r["feed_url"], r["entry_id"], r.get("title", ""), r.get("link", ""),
             r.get("published", ""), r.get("summary", ""), stamp),
        )
        new += cur.rowcount
    con.commit()
    return new


def record_terms(con, records):
    for r in records:
        published = r.get("published") or ""
        try:
            day = day_of(parse_ts(published)) if published else day_of(now_utc())
        except ValueError:
            day = day_of(now_utc())
        for term in set(normalize_terms(r.get("title", ""))):
            con.execute(
                "INSERT OR IGNORE INTO term_counts (term, day, feed_url) VALUES (?,?,?)",
                (term, day, r["feed_url"]),
            )
    con.commit()


def get_feed_cache(con, feed_url):
    row = con.execute(
        "SELECT etag, modified FROM feed_status WHERE feed_url = ?", (feed_url,)
    ).fetchone()
    return (row["etag"], row["modified"]) if row else (None, None)


def set_feed_status(con, feed_url, etag=None, modified=None, error=None):
    stamp = now_utc().isoformat()
    if error is None:
        con.execute(
            "INSERT INTO feed_status (feed_url, etag, modified, last_ok, last_error) "
            "VALUES (?,?,?,?,NULL) "
            "ON CONFLICT(feed_url) DO UPDATE SET etag=excluded.etag, "
            "modified=excluded.modified, last_ok=excluded.last_ok, last_error=NULL",
            (feed_url, etag, modified, stamp),
        )
    else:
        con.execute(
            "INSERT INTO feed_status (feed_url, last_error) VALUES (?,?) "
            "ON CONFLICT(feed_url) DO UPDATE SET last_error=excluded.last_error",
            (feed_url, error),
        )
    con.commit()


def data_span_days(con, now=None):
    row = con.execute("SELECT MIN(day) AS lo, MAX(day) AS hi FROM term_counts").fetchone()
    if not row or not row["lo"]:
        return 0
    lo = parse_ts(row["lo"] + "T00:00:00Z")
    hi = parse_ts(row["hi"] + "T00:00:00Z")
    return int((hi - lo).total_seconds() // 86400)


def was_alerted_recently(con, term, now, cooldown_days):
    row = con.execute(
        "SELECT alerted_at FROM pulse_alerts WHERE term = ?", (term,)
    ).fetchone()
    if not row:
        return False
    return parse_ts(row["alerted_at"]) >= now - timedelta(days=cooldown_days)


def mark_alerted(con, terms, at=None):
    stamp = (at or now_utc()).isoformat()
    con.executemany(
        "INSERT INTO pulse_alerts (term, alerted_at) VALUES (?,?) "
        "ON CONFLICT(term) DO UPDATE SET alerted_at=excluded.alerted_at",
        [(t, stamp) for t in terms],
    )
    con.commit()
```

"""Сбор RSS/Atom-фидов через feedparser → SQLite (+ частоты терминов)."""
import argparse
import sys
from calendar import timegm
from datetime import datetime, timezone

import feedparser

import common


def _entry_date(e):
    t = e.get("published_parsed") or e.get("updated_parsed")
    if not t:
        return ""
    return datetime.fromtimestamp(timegm(t), tz=timezone.utc).isoformat()


def entries_from_parsed(feed_url, parsed):
    records = []
    for e in parsed.entries:
        entry_id = e.get("id") or e.get("link") or e.get("title") or ""
        if not entry_id:
            continue
        records.append({
            "feed_url": feed_url,
            "entry_id": entry_id,
            "title": e.get("title", ""),
            "link": e.get("link", ""),
            "published": _entry_date(e),
            "summary": e.get("summary", ""),
        })
    return records


def _parse(url, etag, modified):
    """Обёртка над feedparser.parse с условным запросом. Отдельно — чтобы мокать."""
    return feedparser.parse(url, etag=etag, modified=modified)


def fetch_feed(con, feed_url):
    etag, modified = common.get_feed_cache(con, feed_url)
    parsed = _parse(feed_url, etag, modified)
    status = getattr(parsed, "status", None)
    if status == 304:                      # не менялся
        common.set_feed_status(con, feed_url,
                               etag=etag, modified=modified)
        return [], None
    if getattr(parsed, "bozo", 0) and not parsed.entries:
        return [], "не удалось разобрать фид"
    records = entries_from_parsed(feed_url, parsed)
    common.set_feed_status(con, feed_url,
                           etag=getattr(parsed, "etag", None),
                           modified=getattr(parsed, "modified", None))
    return records, None


def run_fetch(cfg=None, con=None, limit=None):
    cfg = cfg or common.load_config()
    con = con or common.connect()
    n_new, errors = 0, {}
    for feed_url in cfg["feeds"]:
        try:
            records, err = fetch_feed(con, feed_url)
        except Exception as exc:           # noqa: BLE001 — сеть/парсер
            errors[feed_url] = str(exc)
            common.set_feed_status(con, feed_url, error=str(exc))
            continue
        if err:
            errors[feed_url] = err
            common.set_feed_status(con, feed_url, error=err)
            continue
        if limit:
            records = records[:limit]
        n_new += common.save_entries(con, records)
        common.record_terms(con, records)
    return n_new, errors


def main():
    ap = argparse.ArgumentParser(description="Сбор RSS-фидов в базу rssradar")
    ap.add_argument("--limit", type=int, default=None,
                    help="ограничить число записей на фид (холодный старт)")

    cfg = common.load_config()
    n_new, errors = run_fetch(cfg=cfg, limit=args.limit or cfg["fetch_limit_cold"]
                              if args.limit is None else args.limit)
    print("Новых записей: {}".format(n_new))
    for feed, e in errors.items():
        print("⚠️ {}: {}".format(feed, e), file=sys.stderr)


if __name__ == "__main__":

# discover.py
"""Поиск RSS/Atom-фидов по URL сайта и их валидация."""
import re
from urllib.parse import urljoin, urlparse

import feedparser
import requests

_LINK_RE = re.compile(
    r'<link[^>]+rel=["\']alternate["\'][^>]*>', re.IGNORECASE)
_HREF_RE = re.compile(r'href=["\']([^"\']+)["\']', re.IGNORECASE)
_TYPE_RE = re.compile(r'type=["\']([^"\']+)["\']', re.IGNORECASE)
_COMMON_PATHS = ["/feed", "/rss", "/feed.xml", "/rss.xml", "/atom.xml", "/index.xml"]


def feed_links_in_html(base_url, html):
    found = []
    for tag in _LINK_RE.findall(html or ""):
        tm = _TYPE_RE.search(tag)
        if tm and ("rss" in tm.group(1).lower() or "atom" in tm.group(1).lower()):
            hm = _HREF_RE.search(tag)
            if hm:
                found.append(urljoin(base_url, hm.group(1)))
    root = "{}://{}".format(urlparse(base_url).scheme, urlparse(base_url).netloc)
    for p in _COMMON_PATHS:
        found.append(root + p)
    seen, unique = set(), []
    for u in found:
        if u not in seen:
            seen.add(u)
            unique.append(u)
    return unique


def is_valid_feed(url):
    try:
        parsed = feedparser.parse(url)
    except Exception:                       # noqa: BLE001
        return False
    return bool(parsed.entries) and not (getattr(parsed, "bozo", 0) and not parsed.entries)


def discover(site_url):
    feeds, errors = [], {}
    try:
        resp = requests.get(site_url, timeout=20,
                            headers={"User-Agent": "rssradar/1.0"})
        resp.raise_for_status()
        html = resp.text
    except requests.RequestException as exc:
        html = ""
        errors[site_url] = "сайт недоступен: {}".format(exc)
    for candidate in feed_links_in_html(site_url, html):
        if candidate in feeds:
            continue
        if is_valid_feed(candidate):
            feeds.append(candidate)
    if not feeds and site_url not in errors:
        errors[site_url] = "фид не найден"
    return feeds, errors
```

# telegram.py
"""Отправка отчётов в Telegram ботом пользователя."""
import requests

MAX_LEN = 4000  # лимит API 4096, оставляем запас


def split_message(text, max_len=MAX_LEN):
    if len(text) <= max_len:
        return [text]
    chunks, current = [], ""
    for line in text.split("\n"):
        candidate = current + "\n" + line if current else line
        if len(candidate) > max_len:
            if current:
                chunks.append(current)
            while len(line) > max_len:
                chunks.append(line[:max_len])
                line = line[max_len:]
            current = line
        else:
            current = candidate
    if current:
        chunks.append(current)
    return chunks


def send(text, token, chat_id):
    for chunk in split_message(text):
        resp = requests.post(
            "https://api.telegram.org/bot{}/sendMessage".format(token),
            json={"chat_id": chat_id, "text": chunk,
                  "disable_web_page_preview": True},
            timeout=30,
        )
        resp.raise_for_status()

"""Отчёты rssradar: пульс (всплеск объёма), недельный дайджест, карта."""

import sys
from datetime import timedelta

import common


def term_links(con, term, since_day, limit=3):
    rows = con.execute(
        "SELECT title, link FROM entries WHERE first_seen >= ? "
        "ORDER BY first_seen DESC", (since_day,)
    ).fetchall()
    out = []
    for r in rows:
        if term in common.normalize_terms(r["title"]):
            out.append((r["title"], r["link"]))
        if len(out) >= limit:
            break
    return out


def pulse_signals(con, cfg, now=None):
    now = now or common.now_utc()
    if common.data_span_days(con, now) < cfg["pulse_baseline_days"]:
        return []                                  # базовой линии ещё нет
    window_start = (now - timedelta(days=cfg["pulse_window_days"])).strftime("%Y-%m-%d")
    baseline_start = (now - timedelta(days=cfg["pulse_baseline_days"]
                                      + cfg["pulse_window_days"])).strftime("%Y-%m-%d")
    win_days = max(cfg["pulse_window_days"], 1)
    base_days = max(cfg["pulse_baseline_days"], 1)
    signals = []
    terms = [r["term"] for r in con.execute(
        "SELECT DISTINCT term FROM term_counts WHERE day >= ?", (window_start,))]
    for term in terms:
        recent_feeds = con.execute(
            "SELECT COUNT(DISTINCT feed_url) FROM term_counts "
            "WHERE term = ? AND day >= ?", (term, window_start)).fetchone()[0]
        if recent_feeds < cfg["pulse_min_feeds"]:
            continue
        recent_rows = con.execute(
            "SELECT COUNT(*) FROM term_counts WHERE term = ? AND day >= ?",
            (term, window_start)).fetchone()[0]
        base_rows = con.execute(
            "SELECT COUNT(*) FROM term_counts "
            "WHERE term = ? AND day >= ? AND day < ?",
            (term, baseline_start, window_start)).fetchone()[0]
        recent_daily = recent_rows / win_days
        baseline_daily = base_rows / base_days
        ratio = recent_daily / baseline_daily if baseline_daily else float("inf")
        if ratio < cfg["pulse_spike_multiplier"]:
            continue
        if common.was_alerted_recently(con, term, now, cfg["pulse_cooldown_days"]):
            continue
        signals.append({
            "term": term, "recent_feeds": recent_feeds,
            "recent_daily": recent_daily, "baseline_daily": baseline_daily,
            "ratio": ratio,
            "links": term_links(con, term, window_start),
        })
    signals.sort(key=lambda s: (s["recent_feeds"], s["ratio"]), reverse=True)
    return signals


def format_pulse(signals):
    lines = ["🔥 RSS-пульс: темы набирают обороты", ""]
    for s in signals:
        mult = "∞" if s["ratio"] == float("inf") else "×{:.1f}".format(s["ratio"])
        lines.append("• {} — {} к базовой линии, в {} источниках".format(
            s["term"], mult, s["recent_feeds"]))
        for title, link in s["links"]:
            lines.append("  {}".format(link))
        lines.append("")
    return "\n".join(lines).strip()

import subprocess
from pathlib import Path


def _norm_link(link):
    return (link or "").split("?")[0].rstrip("/").lower()


def weekly_entries(con, cfg, now=None):
    now = now or common.now_utc()
    since = (now - timedelta(days=cfg["weekly_window_days"])).isoformat()
    rows = con.execute(
        "SELECT feed_url, title, link, published, summary, first_seen "
        "FROM entries WHERE first_seen >= ? ORDER BY first_seen DESC", (since,)
    ).fetchall()
    seen, out = set(), []
    for r in rows:
        key = _norm_link(r["link"])
        if key and key in seen:
            continue
        seen.add(key)
        out.append(dict(r))
    return out


def claude_summary(entries, prompt_path):
    """Тематический дайджест недели через headless Claude. Любой сбой → None."""
    if not entries:
        return None
    digest = []
    for e in entries:
        title = " ".join((e["title"] or "").split())
        digest.append("- {} — {}".format(title, e["link"]))
    prompt = Path(prompt_path).read_text(encoding="utf-8") + "\n\n" + "\n".join(digest)
    try:
        result = subprocess.run(
            ["claude", "-p", "--output-format", "text"],
            input=prompt, capture_output=True, text=True, timeout=300,
        )
        if result.returncode == 0 and result.stdout.strip():
            return result.stdout.strip()
    except (OSError, subprocess.TimeoutExpired):
        pass
    return None


def build_weekly(con, cfg, now=None):
    entries = weekly_entries(con, cfg, now=now)
    header = "📰 RSS за неделю"
    if not entries:
        return header + "\n\nЗа неделю новых статей не появилось."
    summary = claude_summary(entries, common.ROOT / "prompts" / "weekly.md")
    if summary:
        return "{}\n\n{}".format(header, summary)
    lines = [header, "", "Новое за неделю:"]           # фолбэк без Claude
    for e in entries:
        title = " ".join((e["title"] or "").split())
        lines.append("• {} — {}".format(title, e["link"]))
    return "\n".join(lines).strip()

```python
import html as html_lib
import json
import webbrowser


def claude_clusters(entries, prompt_path):
    if not entries:
        return None
    digest = ["- {} — {}".format(" ".join((e["title"] or "").split()), e["link"])
              for e in entries]
    prompt = Path(prompt_path).read_text(encoding="utf-8") + "\n\n" + "\n".join(digest)
    try:
        result = subprocess.run(
            ["claude", "-p", "--output-format", "text"],
            input=prompt, capture_output=True, text=True, timeout=300,
        )
        if result.returncode == 0 and result.stdout.strip():
            raw = result.stdout.strip()
            start, end = raw.find("["), raw.rfind("]")
            if start != -1 and end != -1:
                return json.loads(raw[start:end + 1])
    except (OSError, subprocess.TimeoutExpired, json.JSONDecodeError):
        pass
    return None


def render_map_html(clusters, title):
    esc = html_lib.escape
    if not clusters:
        body = "<p class='empty'>Пока нет данных для карты.</p>"
    else:
        cards = []
        for c in clusters:
            heat = max(1, min(int(c.get("heat", 1)), 6))
            items = "".join(
                "<li><a href='{}'>{}</a></li>".format(
                    esc(i.get("link", "")), esc(i.get("title", "")))
                for i in c.get("items", []))
            cards.append(
                "<section class='card heat-{h}'><h2>{t}</h2><ul>{i}</ul></section>".format(
                    h=heat, t=esc(c.get("theme", "")), i=items))
        body = "<div class='grid'>" + "".join(cards) + "</div>"
    return """<!doctype html><html lang='ru'><head><meta charset='utf-8'>
<meta name='viewport' content='width=device-width, initial-scale=1'>
<title>{title}</title><style>
:root{{color-scheme:light dark}}
body{{font:16px/1.5 -apple-system,system-ui,sans-serif;margin:2rem;max-width:1100px}}
h1{{font-size:1.4rem}}
.grid{{display:flex;flex-wrap:wrap;gap:1rem;align-items:flex-start}}
.card{{border:1px solid color-mix(in srgb, currentColor 20%, transparent);
  border-radius:12px;padding:1rem;flex:1 1 240px}}
.card h2{{margin:.2rem 0 .6rem;font-size:1.05rem}}
.heat-6{{flex-basis:420px}} .heat-5{{flex-basis:360px}} .heat-4{{flex-basis:320px}}
.heat-3{{flex-basis:280px}} .heat-2{{flex-basis:240px}} .heat-1{{flex-basis:220px}}
.heat-6 h2::before,.heat-5 h2::before,.heat-4 h2::before{{content:'🔥 '}}
ul{{margin:0;padding-left:1.1rem}} a{{color:inherit}}
.empty{{opacity:.6}}
</style></head><body><h1>{title}</h1>{body}</body></html>""".format(
        title=esc(title), body=body)


def write_and_open(html):
    common.DATA_DIR.mkdir(parents=True, exist_ok=True)
    path = common.DATA_DIR / "map-{}.html".format(common.day_of(common.now_utc()))
    path.write_text(html, encoding="utf-8")
    try:
        webbrowser.open(path.as_uri())
    except Exception:                          # noqa: BLE001 — headless-среда
        pass
    return path


def build_map(con, cfg, now=None):
    entries = weekly_entries(con, cfg, now=now)
    clusters = claude_clusters(entries, common.ROOT / "prompts" / "map.md")
    if clusters is None:                       # фолбэк: одна тема-свалка
        clusters = [{"theme": "Новое за неделю", "heat": 3,
                     "items": [{"title": e["title"], "link": e["link"]}
                               for e in entries]}] if entries else []
    html = render_map_html(clusters, "RSS-карта трендов")
    return str(write_and_open(html))
```

## Онбординг (первый запуск: нет config.json или .env)

Веди по шагам, по одному вопросу за раз, на русском:

1. **Python 3.9+ и зависимости.** `python3 --version`; затем
   `python3 -m pip install -r requirements.txt` (feedparser, requests, pymorphy3).
2. **Источники — два пути:**
   - *Вручную:* пользователь даёт URL сайтов → для каждого
     `python3 -c "import discover, json; print(json.dumps(discover.discover('URL')[0]))"`,
     показать найденные фиды, подтверждённые записать в `config.json` → `feeds`.
   - *По теме:* пользователь называет тему → ты ищешь сайты-кандидаты

     список, подтверждённые фиды пишешь в `config.json`.
3. **Telegram.** Бот: @BotFather → /newbot → токен. chat_id: пользователь пишет
   боту, затем `curl https://api.telegram.org/bot<TOKEN>/getUpdates` →
   `message.chat.id`. Записать в `.env` (`TELEGRAM_TOKEN`, `TELEGRAM_CHAT_ID`).
4. **Холодный старт.** `python3 fetch.py --limit 30` — показать, сколько
   собралось и какие фиды не открылись.
5. **Режим.** Спроси: ручной запуск, недельный дайджест, пульс или всё сразу.
   Для автоматики — `python3 schedule.py install` (предупреди: недельный
